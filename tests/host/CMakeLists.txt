cmake_minimum_required(VERSION 3.16)
project(rebuddy_host C CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CORE "${ROOT}/components/rebuddy_core")
add_library(rebuddy_core STATIC "${CORE}/src/rb_pet.c" "${CORE}/src/rb_choreo.c")
target_include_directories(rebuddy_core PUBLIC "${CORE}/include")
target_compile_options(rebuddy_core PRIVATE -Wall -Wextra -Werror)
target_link_libraries(rebuddy_core PUBLIC m)
enable_testing()
add_executable(rebuddy_board_input_test board_input_test.c
    "${ROOT}/components/rebuddy_board/src/rb_board_input.c")
target_include_directories(rebuddy_board_input_test PRIVATE "${ROOT}/components/rebuddy_board/include")
target_compile_options(rebuddy_board_input_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME board_input COMMAND rebuddy_board_input_test)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME behavior_config COMMAND "${Python3_EXECUTABLE}" "${ROOT}/tools/generate_pet_config.py" --check)
add_test(NAME project_assets COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/project_assets_test.py")
set(REBUDDY_LVGL_DIR "${ROOT}/managed_components/lvgl__lvgl" CACHE PATH "LVGL 9.5 source directory for optional UI tests")
if(EXISTS "${REBUDDY_LVGL_DIR}/lvgl.h")
    find_package(ZLIB REQUIRED)
    set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE PATH "" FORCE)
    set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
    set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    add_subdirectory("${REBUDDY_LVGL_DIR}" lvgl)
    add_executable(rebuddy_ui_test ui_test.c "${ROOT}/main/choreo_ui.c" "${ROOT}/main/choreo_store.c" "${ROOT}/main/pet_ui.c" "${ROOT}/main/pet_assets.c" "${ROOT}/main/pet_animation_table.c" "${ROOT}/main/room_font.c")
    target_include_directories(rebuddy_ui_test PRIVATE "${ROOT}/main")
    target_compile_definitions(rebuddy_ui_test PRIVATE RB_ASSET_FILE="${ROOT}/assets/animation/frames.bin")
    target_compile_definitions(rebuddy_ui_test PRIVATE RB_ROOM_FILE="${ROOT}/assets/animation/room.bin")
    target_link_libraries(rebuddy_ui_test PRIVATE rebuddy_core lvgl ZLIB::ZLIB)
    target_compile_options(rebuddy_ui_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME ui_rgb565 COMMAND rebuddy_ui_test)
    set_tests_properties(ui_rgb565 PROPERTIES TIMEOUT 30)
    add_executable(rebuddy_choreo_ui_test choreo_ui_test.c
        "${ROOT}/main/pet_ui.c" "${ROOT}/main/choreo_ui.c" "${ROOT}/main/choreo_store.c"
        "${ROOT}/main/pet_assets.c" "${ROOT}/main/pet_animation_table.c" "${ROOT}/main/room_font.c")
    target_include_directories(rebuddy_choreo_ui_test PRIVATE "${ROOT}/main")
    target_compile_definitions(rebuddy_choreo_ui_test PRIVATE RB_ASSET_FILE="${ROOT}/assets/animation/frames.bin" RB_ROOM_FILE="${ROOT}/assets/animation/room.bin")
    target_link_libraries(rebuddy_choreo_ui_test PRIVATE rebuddy_core lvgl ZLIB::ZLIB)
    target_compile_options(rebuddy_choreo_ui_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME choreography_ui COMMAND rebuddy_choreo_ui_test)
    set_tests_properties(choreography_ui PROPERTIES TIMEOUT 30)
else()
    message(STATUS "UI tests skipped (LVGL unavailable). Core tests have no SDK or board dependencies.")
endif()

add_executable(rebuddy_pet_test pet_test.c)
target_link_libraries(rebuddy_pet_test PRIVATE rebuddy_core)
target_compile_options(rebuddy_pet_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME reference_behavior COMMAND rebuddy_pet_test)
add_test(NAME animation_resources COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/animation_resources_test.py")
add_test(NAME gif_visual_regressions COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/gifs_v2_test.py")

add_executable(rebuddy_choreo_test choreo_test.c "${ROOT}/main/pet_animation_table.c")
target_include_directories(rebuddy_choreo_test PRIVATE "${ROOT}/main")
target_link_libraries(rebuddy_choreo_test PRIVATE rebuddy_core)
target_compile_options(rebuddy_choreo_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME choreography_model COMMAND rebuddy_choreo_test)
add_executable(rebuddy_choreo_store_test choreo_store_test.c "${ROOT}/main/choreo_store.c")
target_include_directories(rebuddy_choreo_store_test PRIVATE "${ROOT}/main")
target_link_libraries(rebuddy_choreo_store_test PRIVATE rebuddy_core)
target_compile_options(rebuddy_choreo_store_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME choreography_storage COMMAND rebuddy_choreo_store_test)
add_executable(rebuddy_daily_suspend_test core_test.c)
target_link_libraries(rebuddy_daily_suspend_test PRIVATE rebuddy_core)
target_compile_options(rebuddy_daily_suspend_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME daily_suspend COMMAND rebuddy_daily_suspend_test)
add_test(NAME dance_assets COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/dance_assets_test.py")
