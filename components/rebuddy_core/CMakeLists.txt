set(DAILY_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
file(READ "${DAILY_ROOT}/assets/verification/daily_c5_mapping.json" DAILY_REPORT)
foreach(PAIR "record/daily_ready/manifest.json|source_manifest_sha256"
             "components/rebuddy_core/src/rb_daily_data.c|data_sha256"
             "tools/export_daily_c5.py|generator_sha256")
    string(REPLACE "|" ";" PARTS "${PAIR}")
    list(GET PARTS 0 RELATIVE_PATH)
    list(GET PARTS 1 HASH_KEY)
    file(SHA256 "${DAILY_ROOT}/${RELATIVE_PATH}" ACTUAL_HASH)
    string(JSON EXPECTED_HASH GET "${DAILY_REPORT}" "${HASH_KEY}")
    if(NOT ACTUAL_HASH STREQUAL EXPECTED_HASH)
        message(FATAL_ERROR "Stale C5 daily mapping. Run tools/export_daily_c5.py with the RS Python environment.")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${DAILY_ROOT}/${RELATIVE_PATH}")
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${DAILY_ROOT}/assets/verification/daily_c5_mapping.json")
file(READ "${DAILY_ROOT}/assets/verification/dance_c5_mapping.json" DANCE_REPORT)
string(JSON DANCE_HASH GET "${DANCE_REPORT}" data_sha256)
file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/src/rb_dance_data.c" DANCE_ACTUAL)
if(NOT DANCE_HASH STREQUAL DANCE_ACTUAL)
    message(FATAL_ERROR "Stale Dance coefficients; run tools/export_dance_c5.py")
endif()
string(JSON DANCE_SOURCE_COUNT LENGTH "${DANCE_REPORT}" sources)
math(EXPR DANCE_SOURCE_LAST "${DANCE_SOURCE_COUNT}-1")
foreach(I RANGE ${DANCE_SOURCE_LAST})
    string(JSON REL MEMBER "${DANCE_REPORT}" sources ${I})
    string(JSON EXPECTED GET "${DANCE_REPORT}" sources "${REL}")
    file(SHA256 "${DAILY_ROOT}/${REL}" ACTUAL)
    if(NOT ACTUAL STREQUAL EXPECTED)
        message(FATAL_ERROR "Stale Dance source ${REL}; run tools/export_dance_c5.py")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${DAILY_ROOT}/${REL}")
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${DAILY_ROOT}/assets/verification/dance_c5_mapping.json" "${CMAKE_CURRENT_LIST_DIR}/src/rb_dance_data.c")
idf_component_register(SRCS "src/rb_pet.c" "src/rb_choreo.c" "src/rb_motion.c"
    "src/rb_daily.c" "src/rb_daily_data.c" "src/rb_dance.c" "src/rb_dance_data.c" INCLUDE_DIRS "include")
